32]>, pub min_version: u64, } #[derive(Debug, Clone, PartialEq, Eq)] pub enum Gate3Error { EmptyPolicy, KeyNotAllowed, BadKey, BadSignature, MalformedHex, Sha256Mismatch, Blake3Mismatch, ProjectionMismatch, TensorRootMismatch, VersionTooLow, Canon(CanonError), } /// Gate 3: In-toto v1 Statement & Cryptographic Attestation Verifier pub fn verify_gate3( model: &[u8], st: &Statement, signer: &[u8; 32], sig: &[u8; 64], expected_proj: &str, expected_tensor_root: &str, policy: &Policy, ) -> Result<(), Gate3Error> { // 1. Identity Authorization: reject empty policies and unauthorized public keys if policy.allowed_keys.is_empty() { return Err(Gate3Error::EmptyPolicy); } if !policy.allowed_keys.contains(signer) { return Err(Gate3Error::KeyNotAllowed); } // 2.
