//! Gate 4 (apenas o combinador): PENDING nunca vira PASS. Consulta de linhagem/TEE nao implementada offline.
use crate::gate3::Gate3Error;
use crate::gguf::GgufError;
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Gate4Status {
Verified,
Pending(String),
Failed(String),
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Verdict {
Pass,
Reject(String),
Incomplete(String),
}
/// Combinador de veredito fail-closed.
/// AXIOMA DE SEGURANCA: PENDING nunca e PASS.
pub fn final_verdict(
g1: &Result<(), GgufError>,
g3: &Result<(), Gate3Error>,
g4: &Gate4Status,
) -> Verdict {
if let Err(e) = g1 {
return Verdict::Reject(format!("gate1: {:?}", e));
}
if let Err(e) = g3 {
return Verdict::Reject(format!("gate3: {:?}", e));
}
match g4 {
Gate4Status::Verified => Verdict::Pass,
Gate4Status::Pending(r) => Verdict::Incomplete(format!("gate4 pending: {}", r)),
Gate4Status::Failed(r) => Verdict::Reject(format!("gate4: {}", r)),
}
}
