# Arkhe Fail-Closed Gate Spec v0.9.4
name: "Security Gates // Sandbox CVE & WASM Parity"
on:
  pull_request:
    branches: ["canonical-prod"]

jobs:
  enforce-gvisor-version:
    runs-on: ubuntu-latest
    steps:
      - name: Check gVisor Version (CVE-2026-95702 Mitigation)
        run: |
          REQUIRED="20260831.0"
          INSTALLED=$(runsc --version | awk '{print $3}')
          echo "[GATE] Evaluating runsc host binary: $INSTALLED"
          if dpkg --compare-versions "$INSTALLED" "lt" "$REQUIRED"; then
            echo "::error::[CRITICAL FAIL-CLOSED] gVisor $INSTALLED is vulnerable to CVE-2026-95702 (VFS UAF Escape)!"
            exit 1
          fi

      - name: Verify Cosign Keyless Sigstore Attestations
        run: |
          cosign verify-blob \
            --certificate-oidc-issuer="https://token.actions.githubusercontent.com" \
            --certificate-identity-regexp="^https://github.com/arkhe-kernel/*" \
            --bundle=cosign.bundle ./target/release/arkhe-verify-core
          echo "[SIGNATURE] Binary BLAKE3 attestation intact."

  verify-wasm-bit-parity:
    runs-on: ubuntu-latest
    steps:
      - name: Compile x86_64 vs wasm32-wasi & Diff Output

          cargo test -p arkhe-verify-core -- --nocapture

          cargo run -p parity-checker -- tests/golden_vectors.json
